(function() {
  var LIVE_MS = 60000, RETRY_MS = 5000;
  var SENTINEL = 'offgrid-live';
  function schedule(ms) { setTimeout(tick, ms); }
  function tick() {
    var x = new XMLHttpRequest();
    var url = window.location.pathname + '?k=' + (new Date()).getTime();
    x.open('GET', url, true);
    x.onreadystatechange = function() {
      if (x.readyState !== 4) { return; }
      if (x.status !== 200) { schedule(RETRY_MS); return; }
      var t = x.responseText;
      // Markers built by concatenation so this script's own source
      // can never match them (it lives in the fetched head).
      var bo = '<bo' + 'dy';
      var bc = '</bo' + 'dy>';
      var i = t.indexOf(bo);
      if (i < 0) { schedule(RETRY_MS); return; }
      i = t.indexOf('>', i);
      var j = t.lastIndexOf(bc);
      if (i < 0 || j < 0 || j <= i) { schedule(RETRY_MS); return; }
      document.body.innerHTML = t.substring(i + 1, j);
      schedule(t.indexOf(SENTINEL) >= 0 ? LIVE_MS : RETRY_MS);
    };
    x.onerror = function() { schedule(RETRY_MS); };
    x.send(null);
  }
  schedule(LIVE_MS);
})();
</script>
