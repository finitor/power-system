<script type="text/javascript">
(function() {
  var LIVE_MS = 60000, RETRY_MS = 5000;
  var SENTINEL = 'offgrid-live';
  // Pin the footer by computed pixel offset from the measured viewport.
  // This browser ignores min-height/height on the page box (table and
  // block alike) and mishandles position:fixed, so CSS can't place it.
  // Absolute top in px doesn't depend on the container height, so it
  // survives those quirks. Re-run after every body swap.
  function positionFooter() {
    var f = document.getElementById('kindle-footer');
    var d = document.documentElement;
    if (!f || !d || !d.clientHeight) { return; }
    var top = (d.clientHeight - 8) - f.offsetHeight;
    if (top > 0) { f.style.position = 'absolute'; f.style.bottom = 'auto'; f.style.top = top + 'px'; }
    var dg = document.getElementById('vp2');
    if (dg) {
      var kp = document.getElementById('kindle-page');
      dg.innerHTML = 'vp2: vh=' + d.clientHeight + ' footH=' + f.offsetHeight
        + ' top=' + top + ' pageH=' + (kp ? kp.offsetHeight : '?');
    }
  }
  function schedule(ms) { setTimeout(tick, ms); }
  function tick() {
    var x = new XMLHttpRequest();
    var url = window.location.pathname + '?k=' + (new Date()).getTime();
    x.open('GET', url, true);
    x.onreadystatechange = function() {
      if (x.readyState !== 4) { return; }
      if (x.status !== 200) { schedule(RETRY_MS); return; }
      var t = x.responseText;
      // Markers built by concatenation so this script's own source
      // can never match them (it lives in the fetched head).
      var bo = '<bo' + 'dy';
      var bc = '</bo' + 'dy>';
      var i = t.indexOf(bo);
      if (i < 0) { schedule(RETRY_MS); return; }
      i = t.indexOf('>', i);
      var j = t.lastIndexOf(bc);
      if (i < 0 || j < 0 || j <= i) { schedule(RETRY_MS); return; }
      document.body.innerHTML = t.substring(i + 1, j);
      positionFooter();
      schedule(t.indexOf(SENTINEL) >= 0 ? LIVE_MS : RETRY_MS);
    };
    x.onerror = function() { schedule(RETRY_MS); };
    x.send(null);
  }
  setTimeout(positionFooter, 80);
  setTimeout(positionFooter, 600);
  schedule(LIVE_MS);
})();
</script>
