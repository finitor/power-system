<script type="text/javascript">
(function() {
  var LIVE_MS = 60000, RETRY_MS = 5000;
  var FLASH_MS = 900000;
  var SENTINEL = 'offgrid-live';
  var lastFlash = (new Date()).getTime();
  // Full-screen black->white flash forces a full e-ink refresh, clearing
  // ghosting. The overlay is position:fixed, sized by padding to
  // clientHeight (NOT overshot: overshoot extends the document and adds a
  // scrollbar here). It covers the changing content area, which is where
  // ghosting accumulates; the static bottom strip never updates anyway.
  function fullRefresh() {
    var de = document.documentElement;
    var vh = (de && de.clientHeight) || 800;
    var vw = (de && de.clientWidth) || 600;
    var o = document.createElement('div');
    o.style.position = 'fixed';
    o.style.left = '0';
    o.style.top = '0';
    o.style.width = vw + 'px';
    o.style.zIndex = '9999';
    o.style.background = '#000';
    o.style.paddingBottom = vh + 'px';
    document.body.appendChild(o);
    o.offsetHeight;
    setTimeout(function() {
      o.style.background = '#fff';
      o.offsetHeight;
      setTimeout(function() { if (o.parentNode) { o.parentNode.removeChild(o); } }, 400);
    }, 600);
  }
  function schedule(ms) { setTimeout(tick, ms); }
  function tick() {
    var nowMs = (new Date()).getTime();
    var x = new XMLHttpRequest();
    var url = window.location.pathname + '?k=' + nowMs;
    x.open('GET', url, true);
    x.onreadystatechange = function() {
      if (x.readyState !== 4) { return; }
      if (x.status !== 200) { schedule(RETRY_MS); return; }
      var t = x.responseText;
      // Markers built by concatenation so this script's own source
      // can never match them (it lives in the fetched head).
      var bo = '<bo' + 'dy';
      var bc = '</bo' + 'dy>';
      var i = t.indexOf(bo);
      if (i < 0) { schedule(RETRY_MS); return; }
      i = t.indexOf('>', i);
      var j = t.lastIndexOf(bc);
      if (i < 0 || j < 0 || j <= i) { schedule(RETRY_MS); return; }
      document.body.innerHTML = t.substring(i + 1, j);
      if (nowMs - lastFlash >= FLASH_MS) { lastFlash = nowMs; fullRefresh(); }
      schedule(t.indexOf(SENTINEL) >= 0 ? LIVE_MS : RETRY_MS);
    };
    x.onerror = function() { schedule(RETRY_MS); };
    x.send(null);
  }
  schedule(LIVE_MS);
})();
</script>
